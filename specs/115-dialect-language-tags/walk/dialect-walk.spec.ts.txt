async function startSurvey(page: Page): Promise<void> {
  await seedReturningVisitor(page);
  await page.goto("/?e2e=1");
  await page.waitForSelector("#il_language_english", { timeout: 60_000 });
}

async function optionLabels(page: Page): Promise<string[]> {
  const opts = page.getByRole("option");
  const n = await opts.count();
  const labels: string[] = [];
  for (let i = 0; i < n; i++) {
    labels.push((await opts.nth(i).getAttribute("aria-label")) ?? "");
  }
  return labels;
}

/**
 * Pick the LANGUAGE row for `name` — the option whose accessible name is the
 * language itself ("French, français, fr"), not a dialect/variety card
 * ("French, français, fr, American Koine French: a dialect of French (fr) · …")
 * or an uncoded Glottolog row. Scans the rendered options because a /^Name/
 * regex also matches those cards.
 */
async function pickLanguageRow(page: Page, name: string): Promise<string> {
  const opts = page.getByRole("option");
  await expect(opts.first()).toBeVisible({ timeout: 60_000 });
  for (let i = 0; i < 40; i++) {
    const labels = await optionLabels(page);
    const idx = labels.findIndex(
      (l) =>
        l.startsWith(`${name},`) &&
        !/dialect of|variety of|also called|known to Glottolog/.test(l),
    );
    if (idx >= 0) {
      const label = labels[idx]!;
      log(`  pick language row: ${label}`);
      await opts.nth(idx).click();
      return label;
    }
    await page.waitForTimeout(1000);
  }
  throw new Error(
    `no language row for ${name} among: ${JSON.stringify(await optionLabels(page))}`,
  );
}

/** Click the first chooser option whose accessible name matches `re`. */
async function pickOption(page: Page, re: RegExp): Promise<string> {
  const opt = page.getByRole("option", { name: re }).first();
  await expect(opt).toBeVisible({ timeout: 60_000 });
  const label = (await opt.getAttribute("aria-label")) ?? "";
  log(`  pick option: ${label}`);
  await opt.click();
  return label;
}

/**
 * Click the dialect card for `name` ("{name}: a dialect of {language} …").
 * Polls: the "{name} is a dialect: find its language" fallback row matches
 * first while the Glottolog/langtags tiers stream in, and `.first()` would
 * grab it instead of the card the quickstart specifies.
 */
async function pickDialectCard(page: Page, name: string): Promise<string> {
  const opts = page.getByRole("option");
  await expect(opts.first()).toBeVisible({ timeout: 60_000 });
  for (let i = 0; i < 40; i++) {
    const labels = await optionLabels(page);
    const idx = labels.findIndex((l) => new RegExp(`${name}: a (dialect|variety) of`).test(l));
    if (idx >= 0) {
      const label = labels[idx]!;
      log(`  pick dialect card: ${label}`);
      await opts.nth(idx).click();
      return label;
    }
    await page.waitForTimeout(1000);
  }
  throw new Error(
    `no dialect card for ${name} among: ${JSON.stringify(await optionLabels(page))}`,
  );
}

async function typeChooser(page: Page, text: string): Promise<void> {
  await page.locator("#il_language_english").fill(text);
}

/**
 * Drive Q1 (pick `pickRe` after typing `query`) through autonym, code and
 * script, and return the dialect proposal's full tag — or null when no
 * dialect question appears (the layout step renders instead).
 */
async function driveToDialect(
  page: Page,
  opts: {
    query: string;
    pick?: RegExp;
    /** Pick the DIALECT card for this name (not the fallback row). */
    pickDialect?: string;
    /** Pick the LANGUAGE row for this name (not a dialect card). */
    pickLanguage?: string;
    autonym: string;
    script?: string;
  },
): Promise<string | null> {
  await typeChooser(page, opts.query);
  if (opts.pickLanguage !== undefined) {
    await pickLanguageRow(page, opts.pickLanguage);
  } else if (opts.pickDialect !== undefined) {
    await pickDialectCard(page, opts.pickDialect);
  } else {
    await pickOption(page, opts.pick!);
  }
  // The region question sits between English and autonym when the picked
  // language is region-ambiguous (French): advance past it blank here — the
  // region-on-demand test drives it explicitly.
  await passRegionQuestion(page);
  // Q2 autonym (free text is accepted).
  await page.fill("#il_language_autonym", opts.autonym);
  await page.press("#il_language_autonym", "Escape");
  await surveyAdvance(page).click();
  // Q3 code (confirmation of the chooser's result).
  await waitForCodeQuestion(page);
  const codeVal = page.getByTestId("language-code-value");
  if (await codeVal.isVisible().catch(() => false)) {
    log(`  code: ${await codeVal.innerText()}`);
  }
  await surveyAdvance(page).click();
  // Q4 script.
  await page.waitForSelector("#il_target_script", { timeout: 30_000 });
  await selectMenuOption(page, page.locator("#il_target_script"), opts.script ?? "Latn");
  await surveyAdvance(page).click();
  // Dialect question or layout step — poll for whichever lands first.
  const proposal = page.getByTestId("dialect-proposal-tag");
  const layout = page.getByTestId("layout-step");
  for (let i = 0; i < 30; i++) {
    if (await proposal.isVisible().catch(() => false)) {
      const tag = await proposal.innerText();
      log(`  dialect proposal: ${tag}`);
      return tag;
    }
    if (await layout.isVisible().catch(() => false)) {
      log("  no dialect question (layout step)");
      return null;
    }
    await page.waitForTimeout(1000);
  }
  throw new Error("neither dialect proposal nor layout step appeared after script");
}

/**
 * The code question renders two ways: the picker (an input with
 * id="il_language_code") or the confirmation (a <strong
 * data-testid="language-code-value"> with a Change button, no input).
 * Wait for either.
 */
async function waitForCodeQuestion(page: Page): Promise<void> {
  await page
    .locator('#il_language_code, [data-testid="language-code-value"]')
    .first()
    .waitFor({ state: "visible", timeout: 30_000 });
}

/**
 * After a Q1 pick the next question is the region question (when the picked
 * language is region-ambiguous) or the autonym. If the region question
 * appears, pick `regionName` when given (the region the reading already
 * implied) or advance blank (the language's default region).
 */
async function passRegionQuestion(page: Page, regionName?: string): Promise<void> {
  for (let i = 0; i < 30; i++) {
    if (await page.locator("#il_language_region").isVisible().catch(() => false)) {
      if (regionName !== undefined) {
        // The region question is a SelectMenu: open it, then pick by label
        // (its options render only after the trigger is clicked).
        const trigger = page.locator("#il_language_region");
        await trigger.click();
        const opt = page.locator('ul[role="listbox"] li', { hasText: regionName }).first();
        await expect(opt).toBeVisible({ timeout: 10_000 });
        log(`  region question: picked ${regionName}`);
        await opt.click();
      } else {
        log("  region question shown; advancing blank (default region)");
        await surveyAdvance(page).click();
      }
      break;
    }
    if (await page.locator("#il_language_autonym").isVisible().catch(() => false)) break;
    await page.waitForTimeout(1000);
  }
  await page.waitForSelector("#il_language_autonym", { timeout: 30_000 });
}

/** The decisions the survey has recorded, read from localStorage. */
async function decisions(page: Page): Promise<Record<string, unknown>> {
  return page.evaluate(() => {
    const out: Record<string, unknown> = {};
    for (let i = 0; i < localStorage.length; i++) {
      const k = localStorage.key(i);
      if (k === null) continue;
      const v = localStorage.getItem(k) ?? "";
      if (v.includes("language-") || v.includes("dialect") || v.includes("target-script")) {
        out[k] = v.slice(0, 2000);
      }
    }
    return out;
  });
}

async function driveFrenchWithRegion(
  page: Page,
  regionName: string,
): Promise<Record<string, unknown>> {
  await startSurvey(page);
  await typeChooser(page, "French");
  // Typing auto-resolves (no advance): the Customize disclosure appears once
  // the language is chosen. Specify the region BEFORE picking — the pick
  // auto-advances, so Customize would be gone after it. The open listbox
  // covers the Customize button (its <mark> intercepts pointer events), so
  // close it with Escape first — that only closes the popup, it keeps the
  // typed text and the resolved language.
  await page.press("#il_language_english", "Escape");
  const customize = page.getByRole("button", { name: "Customize", exact: true });
  await expect(customize).toBeVisible({ timeout: 30_000 });
  await customize.click({ timeout: 30_000 });
  const regionInput = page.getByLabel("Specify a region…");
  await regionInput.fill(regionName);
  // Escape closed the chooser listbox before Customize, and focus is now in
  // the region field; click the chooser input to focus it, then ArrowDown to
  // open its list (focus alone does not reopen it once closed), before
  // picking the language row.
  await page.click("#il_language_english");
  await page.press("#il_language_english", "ArrowDown");
  await pickLanguageRow(page, "French");
  // The region question follows English once a region was requested. It is a
  // SelectMenu: open it and pick by region code (Canada CA, France FR).
  const regionCode = regionName === "Canada" ? "CA" : "FR";
  await selectMenuOption(page, page.locator("#il_language_region"), regionCode);
  await page.waitForSelector("#il_language_autonym", { timeout: 30_000 });
  await page.fill("#il_language_autonym", "français");
  await page.press("#il_language_autonym", "Escape");
  await surveyAdvance(page).click();
  await waitForCodeQuestion(page);
  await surveyAdvance(page).click();
  await page.waitForSelector("#il_target_script", { timeout: 30_000 });
  await selectMenuOption(page, page.locator("#il_target_script"), "Latn");
  await surveyAdvance(page).click();
  await page.waitForSelector('[data-testid="layout-step"]', { timeout: 30_000 });
  const store = await decisions(page);
  log(`  decisions after ${regionName}: ${JSON.stringify(store)}`);
  return store;
}

/** The composed language tag in the persisted draft. */
async function draftLanguageTag(page: Page): Promise<string | null> {
  return page.evaluate(() => {
    const raw = localStorage.getItem("ks.draft.__pending__.v2");
    if (raw === null) return null;
    try {
      const parsed = JSON.parse(raw) as { languageTag?: unknown };
      return typeof parsed.languageTag === "string" ? parsed.languageTag : null;
    } catch {
      return null;
    }
  });
}

/** Poll the draft's language tag until it reads `expected` (the draft save is debounced). */
async function waitForDraftTag(page: Page, expected: string): Promise<void> {
  await expect
    .poll(async () => draftLanguageTag(page), { timeout: 30_000 })
    .toBe(expected);
}

/** The dialect-related decision records in the persisted draft (probe for walk 4). */
async function dialectDecisions(page: Page): Promise<Record<string, unknown>> {
  return page.evaluate(() => {
    const raw = localStorage.getItem("ks.draft.__pending__.v2");
    if (raw === null) return { probe: "no draft" };
    try {
      const parsed = JSON.parse(raw) as {
        decisions?: Record<string, unknown>;
        surveyAnswers?: Record<string, unknown>;
      };
      const decisions = parsed.decisions ?? {};
      const surveyAnswers = parsed.surveyAnswers ?? {};
      const out: Record<string, unknown> = { decisionKeys: Object.keys(decisions) };
      for (const [k, v] of Object.entries(decisions)) {
        if (k.includes("dialect") || k.includes("language-") || k.includes("target-script")) out[k] = v;
      }
      out.surveyAnswers = surveyAnswers;
      return out;
    } catch (e) {
      return { probe: `parse error: ${String(e)}` };
    }
  });
}

  await page.waitForSelector('[data-testid="layout-step"]', { timeout: 15_000 });

  // <- script <- code <- autonym <- english (5 clicks).
  const back = page.getByRole("button", { name: /^Back$/i });
  for (let i = 0; i < 5; i++) {
    await expect(back).toBeVisible({ timeout: 15_000 });
    await back.click();
    await page.waitForTimeout(500);
  }
  await page.waitForSelector("#il_language_english", { timeout: 15_000 });
  await typeChooser(page, "Catalan");
  await pickLanguageRow(page, "Catalan");
  // Catalan is region-ambiguous (ES/FR): the region question follows the pick.
  await passRegionQuestion(page);
  // Advance back through autonym / code / script to the layout step.
  await page.waitForSelector("#il_language_autonym", { timeout: 30_000 });
  await page.fill("#il_language_autonym", "català");
  await page.press("#il_language_autonym", "Escape");
  await surveyAdvance(page).click();
  await waitForCodeQuestion(page);
  await surveyAdvance(page).click();
  await page.waitForSelector("#il_target_script", { timeout: 30_000 });
  await selectMenuOption(page, page.locator("#il_target_script"), "Latn");
  // The stale gascon answer must be flagged, not silently kept: a footer
  // notice names the step that needs reconfirming. Assert it HERE, still on
  // the identity step — the notice clears on the navigation to layout.
  const beforeStore = await decisions(page);
  log(`  decisions at script step: ${JSON.stringify(beforeStore)}`);
